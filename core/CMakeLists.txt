cmake_minimum_required(VERSION 3.20)
project(kinetic VERSION 0.1.0 LANGUAGES CXX)
add_library(physics STATIC src/physics.cpp)
target_include_directories(physics PUBLIC include)
target_compile_features(physics PUBLIC cxx_std_20)
set_target_properties(physics PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(physics PRIVATE /W4)
else()
  target_compile_options(physics PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(physics-test tests/physics_test.cpp)
target_link_libraries(physics-test PRIVATE physics)
add_test(NAME physics-invariants COMMAND physics-test)
